print(f'Programa para realizar una interpolacion y/o una interpolacion inversa')
n = int(input(f'Ingrese el numero de puntos '))

print(f'Ingrese los puntos tal que (x,y)')
for i in range (0,n):
    x[i],y[i]=float(input()),float(input())

def Grafica(x,y):
    plt.title(f'Grafica de Interpolacion')
    plt.plot(x, y, label='o')
    plt.xlabel('x')
    plt.ylabel('y')
    plt.xlim(x.min()-3,x.max()+3)
    plt.ylim(y.min()-3,y.max()+3)
    #Aqui se ponen los ejes en su lugar con la graduacion

    ax.spines['right'].set_color('none')
    ax.spines['top'].set_color('none')
    ax.xaxis.set_ticks_position('bottom')
    ax.spines['bottom'].set_position(('data',0))
    ax.yaxis.set_ticks_position('left')
    ax.spines['left'].set_position(('data',0))
    #Aqui finaliza
    plt.grid()
    plt.legend()
    plt.show()
    return
def Interpolacion(x,y):
    x1=np.zeros(n,dtype = float)
    y1=np.zeros(n,dtype = float)
    def Casos(x,y,x1,x2,nn,n):
        if(nn==1):
                x1[0],x1[1] = x[0],x[n-1]
                y1[0],y1[1] = y[0],y[n-1]
        elif(nn==2):
            x1[0],x1[1],x1[2] = x[0],x[n-2],x[n-1]
            y1[0],y1[1],y1[2] = y[0],y[n-2],y[n-1]
        elif(nn==3):
            x1[0],x1[1],x1[2],x1[3] = x[0],x[n-3],x[n-2],x[n-1]
            y1[0],y1[1],y1[2],y1[3] = y[0],y[n-3],y[n-2],y[n-1]
        return
    Y = float(0)
    print(f'De acuerdo al grafico de puntos mostrados')
    print(f'Desea hacer un polinomio de lagrange de grado 1, 2, 3... o del grado correspondiente de acuerdo al numero de puntos')
    Nivel = int(input(f'Ingrese el grado que desea manejar o ingrese 0 para el grado correspondiente de acuerdo al numero de puntos '))
    Punto = float(input(f'Ingresa el punto a interpolar '))
    if(Nivel==0): #Grado N
        for i in range (0,n):
            L=float(1)
            for j in range (0,n):
                if(j != i):
                    L = L*((Punto-x[j])/(x[i]-x[j]))
            Y = Y+L*y[i]
        print(f'El punto interpolado tiene coordenada {Punto}, {Y}')
    elif(Nivel==1): #Grado 1
        print(f'Recuerde que debe tener al menos Grado+1 puntos para poder realizar este nivel de polinomio')
        nn = 1
        Casos(x,y,x1,y1,nn,n)
        for i in range (0,nn+1):
            L=float(1)

            for j in range (0,nn+1):
                if(j != i):
                    L = L*((Punto-x1[j])/(x1[i]-x1[j]))
            Y = Y+L*y1[i]
        print(f'El punto interpolado tiene coordenada {Punto}, {Y}')
    elif(Nivel==2): #Grado 2
        print(f'Recuerde que debe tener al menos Grado+1 puntos para poder realizar este nivel de polinomio')
        nn = 2
        Casos(x,y,x1,y1,nn,n)
        for i in range (0,nn+1):
            L=1
            for j in range (0,nn+1):
                if(j != i):
                    L = L*((Punto-x1[j])/(x1[i]-x1[j]))
            Y = Y+L*y1[i]
        print(f'El punto interpolado tiene coordenada {Punto}, {Y}')
    elif(Nivel==3): #Grado 3
        print(f'Recuerde que debe tener al menos Grado+1 puntos para poder realizar este nivel de polinomio')
        nn = 3
        Casos(x,y,x1,y1,nn,n)
        for i in range (0,nn+1):
            L=1
            for j in range (0,nn):
                if(j != i):
                    L = L*((Punto-x1[j])/(x1[i]-x1[j]))
            Y = Y+L*y1[i]
        print(f'El punto interpolado tiene coordenada{Punto}, {Y}')
    return
def InterpolacionExponencial(x,y,n): #A traves de linealizacion
    print(f'En que punto desea hacer la interpolacion exponencial')
    Puntox=float(input())
    LE = np.zeros(5,dtype=float)
    for i in range(0,n):
        LE[0] = LE[0] + x[i]
        LE[1] = LE[1] + math.log10(y[i])
        LE[2] = LE[2] + (x[i]*(math.log10(y[i])))
        LE[3] = LE[3] + (x[i]**2)
    LE[4] = LE[0]**2
    m= ((n*LE[2])-(LE[0]*LE[1]))/((n*LE[3])-LE[4])
    b= (LE[1] - m*LE[0])/n
    r=10**m
    a=10**b
    Puntoy = a*(r**Puntox)
    print(f'El punto interpolado tiene coordenada {Puntox}, {Puntoy}' )
    return
def InterpolacionInversa(x,y):
    Nopt = int(0)
    xp,yp = np.zeros(n,dtype=float),np.zeros(n,dtype=float)
    Puntoy = float(input(f'En que punto en Y, desea hacer la interpolacion inversa '))
    if(n>6): #Para evitar polinomios de interpolacion mayores al grado 5 y asi evitar error
        for i in range(0,n):
            if(y[i]>Puntoy):
                CotaMayor = i
            elif(y[i]<Puntoy):
                CotaMenor = i
        Nopt = CotaMayor-CotaMenor
        for i in range(0,Nopt):
            xp[i] = x[CotaMenor+i]
            yp[i] = y[CotaMenor+i]
    elif(n<=6):
        for i in range(0,n):
            if(y[i]>Puntoy):
                CotaMayor = i
            elif(y[i]<Puntoy):
                CotaMenor = i
        Nopt = int(n)
        xp = x
        yp = y

    Xo = np.zeros((Nopt,Nopt),dtype=float) #Metodo de interpolacion de polinomio unico
    Yo = np.zeros((Nopt,1),dtype=float)
    Cof = np.zeros((Nopt),dtype=float)
    xy=np.zeros((Nopt,2),dtype=float)
    for i in range(0,Nopt):
        xy[i][0]= xp[i]
        xy[i][1] = yp[i]
    for i in range (0,Nopt):
        Yo[i][0]= xy[i][1]
        Xo[i][0] = 1
        Xo[i][1] = xy[i][0]
        for j in range (2,Nopt):
            Xo[i][j] = (xy[i][0])**(j)
    Xo = np.linalg.inv(Xo)
    Cof = np.dot(Xo,Yo)
    for i in range (0,Nopt):
        if(Cof[i]<0.001 and Cof[i]>-0.001): #Para evitar decimales que afecten el resultado
            Cof[i] = 0


    NCof=np.zeros(Nopt,dtype=float)
    for i in range (0,Nopt):
        NCof[i] = Cof[Nopt-i-1]
    NCof[Nopt-1] = NCof[Nopt-1] + (-1*Puntoy) #Se incluye el punto y al final para la raiz

    TFP = np.zeros(9,dtype=float) #Metodo Newton-Rapson
    gr=int(Nopt)
    itera = 0
    errorant=1
    h = 0.01
    error=0.0001
    itermax=50
    TFP[3] = x[CotaMayor]
    while(errorant>error and itera<itermax):
        TFP[4],TFP[6],TFP[7],TFP[8] = 0,0,0,0
        TFP[6] = TFP[3] + h
        for i in range (0,gr):
            if(i == gr-1):
                TFP[4] = TFP[4] + NCof[i]*(TFP[3]**0)
                TFP[7] = TFP[7] + NCof[i]*(TFP[6]**0)
            else:
                TFP[4]= TFP[4] + NCof[i]*(TFP[3]**(gr-i-1))
                TFP[7]= TFP[7] + NCof[i]*(TFP[6]**(gr-i-1))
        TFP[5]=(TFP[7]-TFP[4])/h
        TFP[8] = TFP[3]-(TFP[4]/TFP[5])
        errorant = abs(((TFP[8]-TFP[3])/TFP[6])/100.0)
        itera = itera + 1
        TFP[3] = TFP[8]
    print(f'El punto correspondiente en x para el punto y dado es {TFP[8]}')
    print(f'')
    input(f"Presiona Enter para continuar...")
    return
